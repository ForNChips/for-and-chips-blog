{{- define "main" }} {{/* ───────────────────────────────────────────────────────────── Article graph ───────────────────────────────────────────────────────────── Nodes = published articles. Two edge sets, switchable in the UI: refs : article A links to article B via {{< ref "articles/..." >}} tags : articles share ≥1 tag (edge weight = number of shared tags) */}} {{- $articles := where site.RegularPages "Type" "article" -}} {{/* Build node list. */}} {{- $nodes := slice -}} {{- range $articles -}} {{- $tagList := slice -}} {{- range .Params.tags -}}{{ $tagList = $tagList | append (lower .) }}{{- end -}} {{- $authorNames := slice -}} {{- range .Params.authors -}} {{- $authorNames = $authorNames | append (partial "authors/author_name.html" (printf "%s" (index . "id"))) -}} {{- end -}} {{- $nodes = $nodes | append (dict "id" .RelPermalink "title" .Title "url" .RelPermalink "tags" $tagList "authors" $authorNames "date" (.Date.Format "02 Jan 2006") ) -}} {{- end -}} {{/* Cross-reference edges — parse `{{< ref "articles/..." >}}` from each article's raw content. We keep the regex permissive (accepts ref and relref) and resolve the target via site.GetPage. */}} {{- $refLinks := slice -}} {{- range $articles -}} {{- $source := .RelPermalink -}} {{- $matches := findRE `\{\{<\s*(?:ref|relref)\s+"[^"]+"` .RawContent -}} {{- range $matches -}} {{- $path := replaceRE `^.*"([^"]+)".*$` "$1" . -}} {{- $target := site.GetPage $path -}} {{- if and $target (ne $target.RelPermalink $source) -}} {{- $refLinks = $refLinks | append (dict "source" $source "target" $target.RelPermalink) -}} {{- end -}} {{- end -}} {{- end -}} {{/* Shared-tag edges — pairwise. */}} {{- $tagLinks := slice -}} {{- range $i, $a := $articles -}} {{- $aTags := slice -}} {{- range $a.Params.tags -}}{{ $aTags = $aTags | append (lower .) }}{{- end -}} {{- range $j, $b := $articles -}} {{- if lt $i $j -}} {{- $bTags := slice -}} {{- range $b.Params.tags -}}{{ $bTags = $bTags | append (lower .) }}{{- end -}} {{- $shared := intersect $aTags $bTags -}} {{- if gt (len $shared) 0 -}} {{- $tagLinks = $tagLinks | append (dict "source" $a.RelPermalink "target" $b.RelPermalink "value" (len $shared) ) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
