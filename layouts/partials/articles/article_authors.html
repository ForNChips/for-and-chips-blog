{{- /* Author display with affiliation numbers (shared across authors) */ -}} {{- $allAuthors := hugo.Data.authors }} {{- $postAuthors := .Params.authors }} {{- if $postAuthors }} {{- /* 1. Build the affiliation -> number mapping */ -}} {{- $affIndex := dict }} {{- $counter := 1 }} {{- range $i, $pa := $postAuthors }} {{- range $i, $aff := $pa.affiliations }} {{- if eq (index $affIndex $aff) nil }} {{- $affIndex = merge $affIndex (dict $aff $counter) }} {{- $counter = add $counter 1 }} {{- end }} {{- end }} {{- end }} {{- /* 2. Render authors with affiliation superscripts */ -}}
{{- range $index, $pa := $postAuthors }} {{- $author := index $allAuthors $pa.id }} {{- if $author }} {{- $fullname := printf "%s %s" (upper $author.last_name) $author.first_name -}} {{ $fullname }} {{- range $i, $aff := $pa.affiliations }} {{ index $affIndex $aff }} {{- end }} {{ if lt (add $index 1) (len $postAuthors) }}, {{ end }} {{- else }} {{ $pa.id }}?{{ if lt (add $index 1) (len $postAuthors) }}, {{ end }} {{- end }} {{- end }}
{{- /* 3. Render affiliations sorted by number */ -}}
{{- range $num := seq 1 (len $affIndex) }} {{- range $aff, $val := $affIndex }} {{- if eq $val $num }}
{{ $num }} {{ $aff }}
{{- end }} {{- end }} {{- end }}
{{- end }}
